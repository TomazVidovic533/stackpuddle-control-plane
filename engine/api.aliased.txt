	CommandUp Command = iota + 1

    MarshalText implements encoding.TextMarshaler.

    String returns the wire name, or "" for an invalid value.

func (c *Command) UnmarshalText(text []byte) error
    UnmarshalText implements encoding.TextUnmarshaler and refuses an unknown
    name.

package secrets // import "github.com/TomazVidovic533/stackpuddle-control-plane/internal/secrets"

type Value struct {
	// Has unexported fields.
}
    Value is a secret value. Every print path shows "[redacted]".

func NewValue(b []byte) Value
    NewValue wraps b, which the Value then owns.

func SCRAMSHA256(password Value, random io.Reader) (Value, error)
    SCRAMSHA256 returns the SCRAM-SHA-256 verifier of
    password, in the form a server stores for a role,
    "SCRAM-SHA-256$<iterations>:<salt>$<StoredKey>:<ServerKey>" (RFC 5802,
    RFC 7677). The salt is read from random. The verifier is as secret as the
    password it comes from, so it is a Value.

func (Value) Format(f fmt.State, _ rune)
    Format implements fmt.Formatter so no verb prints the bytes.

func (Value) GoString() string
    GoString implements fmt.GoStringer.

func (v Value) Len() int
    Len returns the length of the value in bytes.

func (Value) LogValue() slog.Value
    LogValue implements slog.LogValuer.

func (Value) MarshalJSON() ([]byte, error)
    MarshalJSON implements json.Marshaler.

func (Value) MarshalText() ([]byte, error)
    MarshalText implements encoding.TextMarshaler.

func (v Value) Reveal() []byte
    Reveal returns the secret bytes. Only the sealed-file writer and the
    child-environment builder call it.

func (Value) String() string
    String implements fmt.Stringer.
